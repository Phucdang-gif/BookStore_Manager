    employees {
        int employee_id PK
        string full_name
        date date_of_birth
        enum gender
        string phone
        string address
        string position
        decimal salary
        date hire_date
        date termination_date
        enum status
        string avatar
    }

    accounts {
        int account_id PK
        int employee_id FK
        int permission_group_id FK
        string username
        string password
        enum status
        datetime created_at
    }

    permission_groups {
        int permission_group_id PK
        string group_name
        enum status
        datetime created_at
    }

    functions {
        int function_id PK
        string function_name
        string system_function_code
        string function_group
    }

    permission_details {
        int detail_id PK
        int permission_group_id FK
        int function_id FK
        string actions
        datetime assigned_at
    }

    customers {
        int customer_id PK
        string full_name
        string phone
        int loyalty_points
        date registration_date
    }

    point_redemption_history {
        int history_id PK
        int customer_id FK
        int points_redeemed
        decimal value_received
        datetime redemption_date
        string redemption_type
    }

    publishers {
        int publisher_id PK
        string publisher_name
        string phone
        enum status
    }

    categories {
        int category_id PK
        string category_name
        int display_order
        enum status
    }

    books {
        int book_id PK
        int publisher_id FK
        int category_id FK
        string isbn
        string book_title
        int page_count
        string language
        int publication_year
        string cover_type
        decimal import_price
        decimal selling_price
        int stock_quantity
        int minimum_stock
        string image
        enum status
        datetime added_at
    }

    authors {
        int author_id PK
        string author_name
    }

    book_authors {
        int book_author_id PK
        int book_id FK
        int author_id FK
        int display_order
    }

    invoices {
        int invoice_id PK
        int customer_id FK
        int employee_id FK
        datetime created_at
        decimal total_amount
        decimal total_discount
        decimal points_used
        decimal points_value
        decimal final_amount
        string payment_method
        string status
        int points_earned
    }

    invoice_details {
        int detail_id PK
        int invoice_id FK
        int book_id FK
        int quantity
        decimal unit_price
        decimal discount
        decimal subtotal
    }

    discount_services {
        int service_id PK
        string service_name
        string discount_type
        decimal discount_value
        decimal minimum_value
        decimal maximum_discount
        datetime start_date
        datetime end_date
        enum status
        text description
    }

    invoice_services {
        int invoice_service_id PK
        int invoice_id FK
        int service_id FK
        string service_type
        decimal discount_value
        text description
    }

    suppliers {
        int supplier_id PK
        string supplier_name
        string phone
        enum status
    }

    import_receipts {
        int receipt_id PK
        int supplier_id FK
        int employee_id FK
        datetime import_date
        datetime receipt_date
        decimal total_amount
        string status
        text notes
    }

    import_receipt_details {
        int detail_id PK
        int receipt_id FK
        int book_id FK
        int quantity
        decimal unit_price
        decimal subtotal
        date expiry_date
    }

    system_parameters {
        string parameter_code PK
        string parameter_value
        string description
